> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** o `teleporte.py` ganha um modo `--ipc` que recebe frames por memória compartilhada e comandos JSON no stdin, para o app SwiftUI (PR 2) controlar o motor.

**Architecture:** `ipc.py` concentra a memória compartilhada (seqlock em dois slots BGRA) e o protocolo JSON por linha. O estado do efeito sai do `main()` para a classe `Teleporter`, usada tanto pelo modo terminal (OpenCV + teclas, sem mudança de uso) quanto pelo `run_ipc()`.

**Tech Stack:** Python 3.12, numpy, OpenCV, MediaPipe 0.10.14, pyvirtualcam, sounddevice; testes com `unittest` da stdlib.

**Spec:** `docs/superpowers/specs/2026-10-24-app-macos-design.md`

## Global Constraints

- Resolução fixa `W, H = 1280, 720`; slots em BGRA (`width*height*4` bytes).
- Cabeçalho de 64 bytes: `magic "TLPT"`, `version u32`, `width u32`, `height u32`, `in_seq u64` (offset 16), `out_seq u64` (offset 24).
- Seqlock: escritor põe `seq` ímpar, copia, põe par; leitor descarta se ímpar ou se mudou.
- Comandos: `{"cmd": ...}` ∈ `start, capture_bg, toggle, reappear, set, test_sound, stop`. Eventos: `{"event": ...}` ∈ `ready, state, gesture_seen, mic_level, error`.
- `state` no máximo 5/s; `mic_level` no máximo 10/s. No modo `--ipc`, o stdout é só protocolo: todo `print` vai para o stderr.
- Modo terminal (`./teleporte`) continua igual: teclas B/R/Q, janela "teleporte".
- Sem dependência nova: testes com `unittest`.
- Testes rodam com `../../.venv/bin/python -m unittest discover -s tests -v` (venv do checkout principal) e precisam de `sons/teleport.wav` (`tools/preparar_som.py`).

```python
# ipc.py — ponte entre o app e o motor: frames por memória compartilhada, comandos e eventos em JSON por linha
import json, mmap, os, struct, sys
import numpy as np

MAGIC, VERSION = b"TLPT", 1
HDR = struct.Struct("<4sIII")  # magic, version, width, height
HDR_SIZE = 64
SEQ_OFF = {"in": 16, "out": 24}
COMMANDS = {"start", "capture_bg", "toggle", "reappear", "set", "test_sound", "stop"}

class FrameShm:
    # dois slots BGRA (in = câmera, out = resultado), cada um com seqlock: seq ímpar = escrevendo
    # ponytail: leitura por polling; semáforo POSIX se o uso de CPU aparecer no perfil
    def __init__(self, path, w, h, create=False):
        size = w * h * 4
        total = HDR_SIZE + 2 * size
        fd = os.open(path, os.O_RDWR | (os.O_CREAT if create else 0), 0o600)
        try:
            if create:
                os.ftruncate(fd, total)
            self.mm = mmap.mmap(fd, total)
        finally:
            os.close(fd)
        if create:
            HDR.pack_into(self.mm, 0, MAGIC, VERSION, w, h)
            struct.pack_into("<QQ", self.mm, SEQ_OFF["in"], 0, 0)
        if HDR.unpack_from(self.mm, 0) != (MAGIC, VERSION, w, h):
            self.mm.close()
            raise ValueError(f"frames.shm incompatível com {w}x{h} v{VERSION}")
        self.view = {s: np.frombuffer(self.mm, np.uint8, size, HDR_SIZE + i * size).reshape(h, w, 4)
                     for i, s in enumerate(("in", "out"))}
        self.last = {"in": 0, "out": 0}

    def _seq(self, slot):
        return struct.unpack_from("<Q", self.mm, SEQ_OFF[slot])[0]

    def _set_seq(self, slot, v):
        struct.pack_into("<Q", self.mm, SEQ_OFF[slot], v)

    def write(self, slot, frame):
        s = self._seq(slot) & ~1  # um escritor por slot: seq par aqui
        self._set_seq(slot, s + 1)
        self.view[slot][...] = frame
        self._set_seq(slot, s + 2)

    def read(self, slot):
        s = self._seq(slot)
        if s & 1 or s == self.last[slot]:
            return None
        frame = self.view[slot].copy()
        if self._seq(slot) != s:
            return None  # o escritor passou por cima durante a cópia
        self.last[slot] = s
        return frame

    def close(self):
        self.view = None  # as views seguram o buffer: soltar antes de fechar
        self.mm.close()

def parse_command(line):
    msg = json.loads(line)
    if not isinstance(msg, dict) or msg.get("cmd") not in COMMANDS:
        raise ValueError(f"comando inválido: {line.strip()[:80]}")
    return msg

def emit(event, out=sys.stdout, **fields):
    out.write(json.dumps({"event": event, **fields}) + "\n")
    out.flush()
```

def open_mic_bridge(mic_name=None, out_names=VIRTUAL_OUTS):
    devs = sd.query_devices()
    out = next((i for n in out_names for i, d in enumerate(devs) if n in d["name"] and d["max_output_channels"] >= 2), None)
    if out is None:
        print("Saída de áudio virtual não encontrada: o som do teleporte toca só aqui, a chamada não ouve.")
        return None
    virtual = lambda d: any(n in d["name"] for n in out_names)
    mic = next((i for i, d in enumerate(devs) if d["name"] == mic_name and d["max_input_channels"] > 0), None) if mic_name else None
    if mic is None:
        mic = sd.default.device[0]
        if mic < 0 or virtual(devs[mic]):  # sem entrada padrão, ou entrada do sistema já na saída virtual
            mic = next((i for i, d in enumerate(devs) if d["max_input_channels"] > 0 and not virtual(d)), None)
    if mic is None:
        print("Nenhum microfone encontrado: a câmera funciona, mas a chamada não recebe áudio deste script.")
        return None
    print(f"Microfone '{devs[mic]['name']}' + teleporte -> '{devs[out]['name']}'. Na chamada, escolha esse microfone.")
    streams = (sd.InputStream(device=mic, samplerate=SR, channels=1, dtype="float32", blocksize=BLOCK, callback=on_mic),
               sd.OutputStream(device=out, samplerate=SR, channels=2, dtype="float32", blocksize=BLOCK, callback=on_audio))
    for s in streams:
        s.start()
    return streams

def run_ipc():
    # modo app: frames entram e saem por memória compartilhada; comandos no stdin, eventos no stdout
    global sfx, sfx_gain
    proto, sys.stdout = sys.stdout, sys.stderr  # print() solto não pode sujar o protocolo
    say = lambda event, **f: emit(event, out=proto, **f)
    cmds = queue.Queue()

    def read_stdin():
        for line in sys.stdin:
            if line.strip():
                try:
                    cmds.put(parse_command(line))
                except ValueError as e:
                    say("error", code="bad_command", msg=str(e))
        cmds.put({"cmd": "stop"})  # o app fechou o pipe (ou morreu): encerra junto

    threading.Thread(target=read_stdin, daemon=True).start()
    say("ready", version=1)
    start = cmds.get()
    while start["cmd"] != "start":
        if start["cmd"] == "stop":
            return 0
        start = cmds.get()
    if not os.path.exists(SOUND):
        say("error", code="no_sound", msg="sons/teleport.wav não encontrado")
        return 1
    sfx, sfx_gain = load_sfx(), float(start.get("gain", SFX_GAIN))
    try:
        shm = FrameShm(start["shm"], W, H)
    except (KeyError, OSError, ValueError) as e:
        say("error", code="bad_shm", msg=str(e))
        return 1
    bridge = None  # guarda a referência: os fluxos de áudio vivem enquanto ela existir
    if start.get("output", "") is not None:
        bridge = open_mic_bridge(start.get("mic"), (start["output"],) if start.get("output") else VIRTUAL_OUTS)
        if bridge is None:
            say("error", code="audio_unavailable", msg="sem saída virtual ou sem microfone")
    vcam = None
    if start.get("vcam", True):
        try:
            vcam = pyvirtualcam.Camera(W, H, fps=30, fmt=pyvirtualcam.PixelFormat.BGR)
        except RuntimeError as e:
            say("error", code="no_virtual_camera", msg=str(e))
    tp = Teleporter(fire_sound)
    tp.reach = float(start.get("reach", REACH))
    next_state = next_level = t0 = time.time()
    frames, seen_sent = 0, False
    try:
        while True:
            now = time.time()
            try:
                c = cmds.get_nowait()
            except queue.Empty:
                c = None
            if c:
                name = c["cmd"]
                if name == "stop":
                    break
                elif name == "capture_bg": tp.capture_bg(now, float(c.get("delay_s", BG_DELAY_S)))
                elif name == "toggle": tp.fire(now)
                elif name == "reappear": tp.reappear()
                elif name == "set":
                    sfx_gain = float(c.get("gain", sfx_gain))
                    tp.reach = float(c.get("reach", tp.reach))
                elif name == "test_sound": fire_sound()
            if now >= next_state:
                say("state", hidden=tp.hidden, bg_ready=tp.bg is not None, bg_countdown=tp.bg_countdown(now),
                    gesture=tp.sign, fps=round(frames / max(now - t0, 1e-6), 1))
                next_state, t0, frames = now + 0.2, now, 0
            if bridge and now >= next_level:
                say("mic_level", rms=round(mic_rms, 4))
                next_level = now + 0.1
            bgra = shm.read("in")
            if bgra is None:
                time.sleep(0.001)
                continue
            out = tp.step(cv2.cvtColor(bgra, cv2.COLOR_BGRA2BGR), now)
            shm.write("out", cv2.cvtColor(out, cv2.COLOR_BGR2BGRA))
            if vcam:
                vcam.send(out)
            frames += 1
            if tp.gesture_seen and not seen_sent:
                say("gesture_seen")
                seen_sent = True
    finally:
        if vcam:
            vcam.close()
        shm.close()
    return 0
