# App SwiftUI, base (PR 2a) — plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** app macOS que captura a webcam, sobe o motor `--ipc`, mostra a prévia processada e controla o teleporte pela tela Início.

**Architecture:** `FrameBuffer` mapeia o `frames.shm` (layout de `ipc.py`) e escreve com seqlock e barreiras de memória; `CameraCapture` (AVFoundation) escreve cada frame no slot `in`; `EngineClient` sobe `teleporte.py --ipc` e troca JSON por linha; `AppModel` (`@Observable`) orquestra e lê o slot `out` a 15 fps para a prévia; `HomeView` em Liquid Glass.

**Tech Stack:** Swift (modo de linguagem 5, compilador 6.4), SwiftUI macOS 26 (`glassEffect`, `.glass`/`.glassProminent`), AVFoundation, Synchronization (`atomicMemoryFence`), XCTest, XcodeGen.

**Spec:** `docs/superpowers/specs/2026-10-24-app-macos-design.md`

## Global Constraints

- macOS 26+, só Apple Silicon (`arm64`). Bundle id `com.mvtiburcio.teleporte`.
- Frames 1280x720 BGRA; cabeçalho de 64 bytes `TLPT`, versão 1, `in_seq` no offset 16, `out_seq` no 24 (igual ao `ipc.py`).
- Comandos `{"cmd": ...}` e eventos `{"event": ...}` como na spec; `start` manda `shm` (e, nos testes, `output: null`, `vcam: false`).
- Câmeras virtuais ("OBS Virtual Camera", "Teleporte Camera") nunca aparecem como fonte (evita realimentação).
- O `.xcodeproj` é gerado (`xcodegen generate`) e não é versionado. Team ID e identidade de assinatura ficam em `app/Local.xcconfig` (fora do git); sem ele, assinatura ad-hoc.
- Caminho do repositório só no Info.plist de Debug (`TELEPORTE_REPO_DIR`); no Release fica vazio.
- Textos da interface em português.
- Build e testes: `cd app && xcodegen generate && xcodebuild test -project Teleporte.xcodeproj -scheme Teleporte -destination 'platform=macOS,arch=arm64' -quiet`.

```swift
import Foundation
import Synchronization

/// Memória compartilhada de frames com o motor (mesmo layout do ipc.py): cabeçalho de 64 bytes
/// e dois slots BGRA (in = câmera, out = resultado), cada um com seqlock: seq ímpar = escrevendo.
final class FrameBuffer: @unchecked Sendable {
    enum Slot {
        case input, output
        var seqOffset: Int { self == .input ? 16 : 24 }
    }

    static let headerSize = 64
    let width: Int, height: Int, slotSize: Int
    private let base: UnsafeMutableRawPointer
    private let total: Int

    init(path: String, width: Int, height: Int) throws {
        self.width = width
        self.height = height
        slotSize = width * height * 4
        total = Self.headerSize + 2 * slotSize
        let fd = open(path, O_RDWR | O_CREAT | O_TRUNC, 0o600)
        guard fd >= 0 else { throw POSIXError(POSIXErrorCode(rawValue: errno) ?? .EIO) }
        defer { close(fd) }
        guard ftruncate(fd, off_t(total)) == 0 else { throw POSIXError(POSIXErrorCode(rawValue: errno) ?? .EIO) }
        let p = mmap(nil, total, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0)
        guard let p, p != MAP_FAILED else { throw POSIXError(POSIXErrorCode(rawValue: errno) ?? .EIO) }
        base = p
        base.copyMemory(from: Array("TLPT".utf8), byteCount: 4)
        for (offset, value) in [(4, 1), (8, width), (12, height)] {
            base.storeBytes(of: UInt32(value).littleEndian, toByteOffset: offset, as: UInt32.self)
        }  // os seqs já nascem zerados pelo O_TRUNC + ftruncate
    }

    deinit { munmap(base, total) }

    private func seq(_ s: Slot) -> UInt64 { base.load(fromByteOffset: s.seqOffset, as: UInt64.self) }
    private func setSeq(_ s: Slot, _ v: UInt64) { base.storeBytes(of: v, toByteOffset: s.seqOffset, as: UInt64.self) }
    private func slot(_ s: Slot) -> UnsafeMutableRawPointer { base + Self.headerSize + (s == .input ? 0 : slotSize) }

    /// Copia um frame BGRA para o slot; bytesPerRow pode ser maior que width*4 (linhas alinhadas).
    func write(_ s: Slot, from src: UnsafeRawPointer, bytesPerRow: Int) {
        let v = seq(s) & ~1  // um escritor por slot
        setSeq(s, v + 1)
        atomicMemoryFence(ordering: .sequentiallyConsistent)
        let dst = slot(s), row = width * 4
        if bytesPerRow == row {
            dst.copyMemory(from: src, byteCount: slotSize)
        } else {
            for y in 0..<height { (dst + y * row).copyMemory(from: src + y * bytesPerRow, byteCount: row) }
        }
        atomicMemoryFence(ordering: .sequentiallyConsistent)
        setSeq(s, v + 2)
    }

    /// Copia o frame novo do slot para dst. Falso se não há frame novo ou se ele foi rasgado durante a cópia.
    func read(_ s: Slot, last: inout UInt64, into dst: UnsafeMutableRawPointer) -> Bool {
        let v = seq(s)
        atomicMemoryFence(ordering: .acquiring)
        guard v & 1 == 0, v != last else { return false }
        dst.copyMemory(from: slot(s), byteCount: slotSize)
        atomicMemoryFence(ordering: .acquiring)
        guard seq(s) == v else { return false }
        last = v
        return true
    }
}
```

```swift
import Foundation

/// Evento do motor (uma linha JSON do stdout). Campos ausentes ficam nil.
struct EngineEvent: Decodable, Sendable {
    let event: String
    var version: Int?
    var hidden: Bool?
    var bgReady: Bool?
    var bgCountdown: Double?
    var gesture: Bool?
    var fps: Double?
    var rms: Double?
    var code: String?
    var msg: String?

    static func parse(_ line: String) -> EngineEvent? {
        let d = JSONDecoder()
        d.keyDecodingStrategy = .convertFromSnakeCase
        return try? d.decode(EngineEvent.self, from: Data(line.utf8))
    }
}

/// Sobe o motor (teleporte.py --ipc) como processo filho: comandos no stdin, eventos no stdout, stderr no log.
final class EngineClient: @unchecked Sendable {
    var onEvent: @MainActor (EngineEvent) -> Void = { _ in }
    var onExit: @MainActor (Int32) -> Void = { _ in }
    private let process = Process()
    private let input = Pipe(), output = Pipe()
    private var pending = Data()  // só mexido na fila do readabilityHandler

    /// Executável e argumentos do motor: empacotado no app (release) ou o .venv do repositório (desenvolvimento).
    static func command(repoDir: String? = Bundle.main.object(forInfoDictionaryKey: "TeleporteRepoDir") as? String) -> (URL, [String])? {
        if let bundled = Bundle.main.url(forResource: "teleporte-engine", withExtension: nil, subdirectory: "engine") {
            return (bundled, ["--ipc"])
        }
        guard let repoDir, !repoDir.isEmpty else { return nil }
        let repo = URL(fileURLWithPath: repoDir).standardizedFileURL
        // numa worktree (.worktrees/x) o .venv fica no checkout principal
        // standardizar só a pasta: com "..", o standardizedFileURL resolve symlinks e o .venv/bin/python
        // viraria o python do Homebrew, fora do venv (sem numpy)
        let python = [repo, repo.appendingPathComponent("../..").standardizedFileURL]
            .map { $0.appendingPathComponent(".venv/bin/python") }
            .first { FileManager.default.isExecutableFile(atPath: $0.path) }
        guard let python else { return nil }
        return (python, [repo.appendingPathComponent("teleporte.py").path, "--ipc"])
    }

    func start(executable: URL, arguments: [String], log: URL) throws {
        try FileManager.default.createDirectory(at: log.deletingLastPathComponent(), withIntermediateDirectories: true)
        if !FileManager.default.fileExists(atPath: log.path) { FileManager.default.createFile(atPath: log.path, contents: nil) }
        let logHandle = try FileHandle(forWritingTo: log)
        logHandle.seekToEndOfFile()
        process.executableURL = executable
        process.arguments = arguments
        process.standardInput = input
        process.standardOutput = output
        process.standardError = logHandle
        output.fileHandleForReading.readabilityHandler = { [weak self] handle in
            let data = handle.availableData
            guard let self, !data.isEmpty else { return }
            pending.append(data)
            while let nl = pending.firstIndex(of: 0x0A) {
                let line = String(decoding: pending[pending.startIndex..<nl], as: UTF8.self)
                pending.removeSubrange(pending.startIndex...nl)
                guard let ev = EngineEvent.parse(line) else { continue }
                DispatchQueue.main.async { MainActor.assumeIsolated { self.onEvent(ev) } }
            }
        }
        process.terminationHandler = { [weak self] p in
            DispatchQueue.main.async { MainActor.assumeIsolated { self?.onExit(p.terminationStatus) } }
        }
        try process.run()
    }

    func send(_ cmd: String, _ fields: [String: Any] = [:]) {
        var msg = fields
        msg["cmd"] = cmd
        guard process.isRunning, let data = try? JSONSerialization.data(withJSONObject: msg) else { return }
        try? input.fileHandleForWriting.write(contentsOf: data + Data([0x0A]))  // motor morto: EPIPE vira erro ignorado
    }

    /// Pede pra sair; se em 2 s o motor não saiu, encerra à força.
    func stop() {
        send("stop")
        DispatchQueue.global().asyncAfter(deadline: .now() + 2) { [process] in
            if process.isRunning { process.terminate() }
        }
    }
}
```

```swift
import AVFoundation

enum CaptureError: LocalizedError {
    case notFound, unsupported
    var errorDescription: String? {
        switch self {
        case .notFound: "Câmera não encontrada."
        case .unsupported: "Esta câmera não aceita o formato de vídeo do Teleporte."
        }
    }
}

/// Captura a webcam escolhida em BGRA no tamanho do FrameBuffer e escreve cada frame no slot in.
final class CameraCapture: NSObject, AVCaptureVideoDataOutputSampleBufferDelegate, @unchecked Sendable {
    static let virtualNames = ["OBS Virtual Camera", "Teleporte Camera"]  // nunca capturar a própria saída
    private let session = AVCaptureSession()
    private let queue = DispatchQueue(label: "teleporte.camera")
    private let buffer: FrameBuffer

    init(buffer: FrameBuffer) {
        self.buffer = buffer
    }

    static func cameras() -> [AVCaptureDevice] {
        AVCaptureDevice.DiscoverySession(deviceTypes: [.builtInWideAngleCamera, .external, .continuityCamera],
                                         mediaType: .video, position: .unspecified)
            .devices.filter { d in !virtualNames.contains { d.localizedName.contains($0) } }
    }

    func start(deviceID: String) throws {
        guard let device = AVCaptureDevice(uniqueID: deviceID) else { throw CaptureError.notFound }
        let input = try AVCaptureDeviceInput(device: device)
        let output = AVCaptureVideoDataOutput()
        output.videoSettings = [  // o próprio AVFoundation escala pro tamanho do motor
            kCVPixelBufferPixelFormatTypeKey as String: kCVPixelFormatType_32BGRA,
            kCVPixelBufferWidthKey as String: buffer.width,
            kCVPixelBufferHeightKey as String: buffer.height,
        ]
        output.alwaysDiscardsLateVideoFrames = true
        output.setSampleBufferDelegate(self, queue: queue)
        session.beginConfiguration()
        defer { session.commitConfiguration() }
        guard session.canAddInput(input), session.canAddOutput(output) else { throw CaptureError.unsupported }
        session.addInput(input)
        session.addOutput(output)
        queue.async { self.session.startRunning() }
    }

    func stop() {
        queue.async { self.session.stopRunning() }
    }

    func captureOutput(_ output: AVCaptureOutput, didOutput sampleBuffer: CMSampleBuffer, from connection: AVCaptureConnection) {
        guard let px = CMSampleBufferGetImageBuffer(sampleBuffer) else { return }
        CVPixelBufferLockBaseAddress(px, .readOnly)
        defer { CVPixelBufferUnlockBaseAddress(px, .readOnly) }
        guard CVPixelBufferGetWidth(px) == buffer.width, CVPixelBufferGetHeight(px) == buffer.height,
              let src = CVPixelBufferGetBaseAddress(px) else { return }
        buffer.write(.input, from: src, bytesPerRow: CVPixelBufferGetBytesPerRow(px))
    }
}
```

O `startRunning` vai para a fila depois do `commitConfiguration` porque o `defer` roda ao sair do `start`, antes do bloco assíncrono executar.

```swift
import AVFoundation
import Observation
import SwiftUI

/// Estado do app: liga câmera + motor, repassa comandos e mantém a prévia (slot out a 15 fps).
@MainActor @Observable
final class AppModel {
    static let width = 1280, height = 720
    static let supportDir = FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0]
        .appendingPathComponent("Teleporte")
    static let logURL = FileManager.default.urls(for: .libraryDirectory, in: .userDomainMask)[0]
        .appendingPathComponent("Logs/Teleporte/engine.log")

    var running = false
    var state: EngineEvent?
    var preview: CGImage?
    var lastError: String?
    var cameraID: String?

    @ObservationIgnored private var engine: EngineClient?
    @ObservationIgnored private var capture: CameraCapture?
    @ObservationIgnored private var frames: FrameBuffer?
    @ObservationIgnored private var timer: Timer?
    @ObservationIgnored private var lastOut: UInt64 = 0
    @ObservationIgnored private var outBytes = [UInt8](repeating: 0, count: width * height * 4)

    func toggleRunning() async {
        if running { stop() } else { await start() }
    }

    func start() async {
        lastError = nil
        guard await AVCaptureDevice.requestAccess(for: .video) else {
            lastError = "Sem permissão de câmera: libere em Ajustes > Privacidade e Segurança > Câmera."
            return
        }
        guard let id = cameraID ?? CameraCapture.cameras().first?.uniqueID else {
            lastError = "Nenhuma câmera encontrada."
            return
        }
        guard let (exe, args) = EngineClient.command() else {
            lastError = "Motor não encontrado: rode ./setup.sh no repositório."
            return
        }
        do {
            try FileManager.default.createDirectory(at: Self.supportDir, withIntermediateDirectories: true)
            let shm = Self.supportDir.appendingPathComponent("frames.shm").path
            let frames = try FrameBuffer(path: shm, width: Self.width, height: Self.height)
            let capture = CameraCapture(buffer: frames)
            try capture.start(deviceID: id)
            let engine = EngineClient()
            let current = ObjectIdentifier(engine)
            engine.onEvent = { [weak self] in self?.handle($0) }
            engine.onExit = { [weak self] code in
                guard let self, let e = self.engine, ObjectIdentifier(e) == current else { return }  // motor antigo saindo
                stop()
                lastError = "O motor parou (código \(code)). Detalhes em \(Self.logURL.path)."
            }
            try engine.start(executable: exe, arguments: args, log: Self.logURL)
            engine.send("start", ["shm": shm])
            (self.frames, self.capture, self.engine, cameraID) = (frames, capture, engine, id)
            running = true
            lastOut = 0
            timer = Timer.scheduledTimer(withTimeInterval: 1.0 / 15, repeats: true) { [weak self] _ in
                MainActor.assumeIsolated { self?.refreshPreview() }
            }
        } catch {
            capture?.stop()
            lastError = error.localizedDescription
        }
    }

    func stop() {
        timer?.invalidate()
        engine?.stop()
        capture?.stop()
        (timer, engine, capture, frames) = (nil, nil, nil, nil)
        running = false
        state = nil
        preview = nil
    }

    func send(_ cmd: String, _ fields: [String: Any] = [:]) {
        engine?.send(cmd, fields)
    }

    private func handle(_ ev: EngineEvent) {
        switch ev.event {
        case "state": state = ev
        case "error": lastError = ev.msg ?? ev.code
        default: break
        }
    }

    private func refreshPreview() {
        guard let frames else { return }
        let ok = outBytes.withUnsafeMutableBytes { frames.read(.output, last: &lastOut, into: $0.baseAddress!) }
        if ok { preview = Self.image(outBytes, width: Self.width, height: Self.height) }
    }

    nonisolated static func image(_ bytes: [UInt8], width: Int, height: Int) -> CGImage? {
        guard let provider = CGDataProvider(data: Data(bytes) as CFData) else { return nil }
        let info = CGBitmapInfo(rawValue: CGImageAlphaInfo.noneSkipFirst.rawValue | CGBitmapInfo.byteOrder32Little.rawValue)
        return CGImage(width: width, height: height, bitsPerComponent: 8, bitsPerPixel: 32, bytesPerRow: width * 4,
                       space: CGColorSpaceCreateDeviceRGB(), bitmapInfo: info, provider: provider,
                       decode: nil, shouldInterpolate: true, intent: .defaultIntent)
    }
}
```

```swift
import SwiftUI

struct HomeView: View {
    @Environment(AppModel.self) private var model

    var body: some View {
        VStack(spacing: 18) {
            preview
            GlassEffectContainer(spacing: 12) {
                HStack(spacing: 12) {
                    Button(model.running ? "Desligar" : "Ligar teleporte", systemImage: model.running ? "stop.fill" : "sparkles") {
                        Task { await model.toggleRunning() }
                    }
                    .buttonStyle(.glassProminent)
                    .tint(.orange)
                    Button(bgLabel, systemImage: "photo") { model.send("capture_bg", ["delay_s": 5]) }
                        .buttonStyle(.glass)
                        .disabled(!model.running)
                    Button("Some / Volta", systemImage: "wand.and.rays") { model.send("toggle") }
                        .buttonStyle(.glass)
                        .disabled(!(model.state?.bgReady ?? false))
                }
                .controlSize(.large)
            }
            status
            if let err = model.lastError {
                Label(err, systemImage: "exclamationmark.triangle.fill")
                    .foregroundStyle(.red)
                    .textSelection(.enabled)
            }
        }
        .padding(24)
        .navigationTitle("Início")
    }

    private var bgLabel: String {
        if let c = model.state?.bgCountdown { return "Fundo em \(Int(c.rounded(.up))) s" }
        return (model.state?.bgReady ?? false) ? "Regravar fundo" : "Gravar fundo"
    }

    private var preview: some View {
        ZStack {
            if let img = model.preview {
                Image(decorative: img, scale: 1)
                    .resizable()
                    .scaleEffect(x: -1, y: 1)  // espelhada, como num espelho; a chamada recebe sem espelhar
            } else {
                Text(model.running ? "Iniciando…" : "Ligue o teleporte para ver a prévia")
                    .foregroundStyle(.secondary)
            }
        }
        .frame(maxWidth: .infinity, maxHeight: .infinity)
        .aspectRatio(16 / 9, contentMode: .fit)
        .clipShape(.rect(cornerRadius: 22))
        .glassEffect(.regular, in: .rect(cornerRadius: 22))
    }

    private var status: some View {
        HStack(spacing: 12) {
            pill("Motor", on: model.state != nil)
            pill("Fundo", on: model.state?.bgReady ?? false)
            pill("Gesto", on: model.state?.gesture ?? false)
            if let fps = model.state?.fps {
                Text("\(Int(fps)) fps").monospacedDigit().foregroundStyle(.secondary)
            }
        }
        .font(.callout)
    }

    private func pill(_ title: String, on: Bool) -> some View {
        Label {
            Text(title)
        } icon: {
            Circle().fill(on ? Color.green : Color.secondary.opacity(0.4)).frame(width: 8, height: 8)
        }
        .padding(.horizontal, 12)
        .padding(.vertical, 6)
        .glassEffect(.regular, in: .capsule)
    }
}
```

```swift
import SwiftUI

struct ContentView: View {
    enum Section: String, CaseIterable, Identifiable {
        case home = "Início"
        var id: Self { self }
        var icon: String {
            switch self {
            case .home: "play.circle"
            }
        }
    }

    @State private var section: Section? = .home

    var body: some View {
        NavigationSplitView {
            List(Section.allCases, selection: $section) { s in
                Label(s.rawValue, systemImage: s.icon)
            }
            .navigationSplitViewColumnWidth(min: 170, ideal: 190)
        } detail: {
            switch section ?? .home {
            case .home: HomeView()
            }
        }
        .frame(minWidth: 820, minHeight: 560)
    }
}
```

```swift
import SwiftUI

@main
struct TeleporteApp: App {
    @State private var model = AppModel()
    @NSApplicationDelegateAdaptor private var delegate: AppDelegate

    init() {
        signal(SIGPIPE, SIG_IGN)  // escrever pra um motor que morreu não pode derrubar o app
    }

    var body: some Scene {
        WindowGroup("Teleporte") {
            ContentView()
                .environment(model)
                .onAppear { delegate.model = model }
        }
    }
}

final class AppDelegate: NSObject, NSApplicationDelegate {
    var model: AppModel?

    func applicationShouldTerminateAfterLastWindowClosed(_ sender: NSApplication) -> Bool { true }

    func applicationWillTerminate(_ notification: Notification) {
        MainActor.assumeIsolated { model?.stop() }  // o motor também sai sozinho quando o stdin fecha
    }
}
```
